class Ticket
    private attributes:
        movieName
        seatNumber
        price
        isBooked

    constructor(movieName, seatNumber, price):
        this.movieName = movieName
        this.seatNumber = seatNumber
        this.price = price
        this.isBooked = true

    method cancelTicket():
        if isBooked == true then
            isBooked = false
            print "Ticket for seat " + seatNumber + " cancelled."
        else
            print "Ticket already cancelled."
class Theatre
    private attributes:
        theatreName
        availableSeats (list)
        bookedTickets (list)

    constructor(theatreName, availableSeats):
        this.theatreName = theatreName
        this.availableSeats = availableSeats
        this.bookedTickets = empty list

    method issueTicket(movieName, seatNumber, price):
        if seatNumber in availableSeats then
            ticket = new Ticket(movieName, seatNumber, price)
            add ticket to bookedTickets
            remove seatNumber from availableSeats
            print "Ticket issued for " + movieName + ", Seat: " + seatNumber
            return ticket
        else
            print "Seat " + seatNumber + " not available."
            return null

    method cancelTicket(ticket):
        if ticket != null and ticket.isBooked == true then
            ticket.cancelTicket()
            add ticket.seatNumber back to availableSeats
            remove ticket from bookedTickets
        else
            print "Invalid or already cancelled ticket."
